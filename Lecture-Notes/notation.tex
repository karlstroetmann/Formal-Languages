% Entries of the list of notation (printed after the table of contents).
%
% Each entry is referenced by \glsadd{key} at the place where the notation is
% defined; \glsadd records the page without printing anything.  The list is
% printed in order of first use, i.e. in the order of the chapters.

% --- Strings and languages ---
\newglossaryentry{alphabet}{name={$\Sigma$}, description={alphabet}}
\newglossaryentry{strings}{name={$\Sigma^*$}, description={set of all strings over the alphabet $\Sigma$}}
\newglossaryentry{lambda}{name={$\lambda$}, description={empty string}}
\newglossaryentry{length}{name={$|w|$}, description={length of the string $w$}}
\newglossaryentry{char}{name={$w[i]$}, description={$i$-th character of the string $w$ (counting from $0$)}}
\newglossaryentry{product}{name={$L_1 \cdot L_2$}, description={product (concatenation) of the languages $L_1$ and $L_2$}}
\newglossaryentry{langpower}{name={$L^n$}, description={$n$-th power of the language $L$}}
\newglossaryentry{kleene}{name={$L^*$}, description={Kleene closure of the language $L$}}
\newglossaryentry{strpower}{name={$s^n$}, description={$n$-th power of the string $s$}}
\newglossaryentry{powerset}{name={$2^M$}, description={power set of the set $M$}}

% --- Regular expressions ---
\newglossaryentry{regexp}{name={$\texttt{RegExp}_\Sigma$}, description={set of regular expressions over $\Sigma$}}
\newglossaryentry{langofre}{name={$L(r)$}, description={language described by the regular expression $r$}}
\newglossaryentry{emptyre}{name={$\emptyset$}, description={regular expression denoting the empty language}}
\newglossaryentry{epsilon}{name={$\varepsilon$}, description={regular expression denoting $\{\lambda\}$}}
\newglossaryentry{choice}{name={$r_1 + r_2$}, description={choice between the regular expressions $r_1$ and $r_2$}}
\newglossaryentry{concat}{name={$r_1 \cdot r_2$}, description={concatenation of the regular expressions $r_1$ and $r_2$}}
\newglossaryentry{star}{name={$r^*$}, description={Kleene star of the regular expression $r$}}
\newglossaryentry{doteq}{name={$r_1 \doteq r_2$}, description={$r_1$ and $r_2$ are equivalent, i.e.~$L(r_1) = L(r_2)$}}

% --- Finite state machines ---
\newglossaryentry{dfa}{name={$\langle Q, \Sigma, \delta, q_0, A \rangle$}, description={finite state machine: states, alphabet, transition function, start state, accepting states}}
\newglossaryentry{omega}{name={$\Omega$}, description={undefined value, e.g.~a missing transition}}
\newglossaryentry{deltastar}{name={$\delta^*(q, w)$}, description={state reached from $q$ after reading the string $w$}}
\newglossaryentry{langoffsm}{name={$L(F)$}, description={language accepted by the automaton $F$}}
\newglossaryentry{skull}{name={$\skull$}, description={dead state}}
\newglossaryentry{leadsto}{name={$\leadsto$}, description={transition between configurations of an \simtextsc{Nfa}}}
\newglossaryentry{leadstostar}{name={$\leadsto^*$}, description={reflexive-transitive closure of $\leadsto$}}
\newglossaryentry{ec}{name={$\textsl{ec}(q)$}, description={$\varepsilon$-closure of the state $q$}}
\newglossaryentry{det}{name={$\textsl{det}(F)$}, description={deterministic automaton constructed from the \simtextsc{Nfa} $F$}}
\newglossaryentry{sim}{name={$p_1 \sim p_2$}, description={the states $p_1$ and $p_2$ are equivalent}}
\newglossaryentry{eqclass}{name={$[p]_\sim$}, description={equivalence class of the state $p$}}

% --- Regular languages ---
\newglossaryentry{complement}{name={$\Sigma^* \backslash L$}, description={complement of the language $L$}}
\newglossaryentry{reversal}{name={$s^R$, $L^R$}, description={reversal of the string $s$, of the language $L$}}

% --- Context-free grammars ---
\newglossaryentry{grammar}{name={$\langle V, T, R, S \rangle$}, description={context-free grammar: variables, terminals, rules, start symbol}}
\newglossaryentry{derives}{name={$\alpha \Rightarrow_G \beta$}, description={$\beta$ results from $\alpha$ by one derivation step of $G$}}
\newglossaryentry{derivesstar}{name={$\alpha \Rightarrow^*_G \beta$}, description={$\beta$ can be derived from $\alpha$ in any number of steps}}
\newglossaryentry{langofg}{name={$L(G)$}, description={language generated by the grammar $G$}}

% --- Parsing ---
\newglossaryentry{earleyobj}{name={$\langle a \rightarrow \beta \bullet \gamma, k \rangle$}, description={Earley object}}
\newglossaryentry{newstart}{name={$\widehat{s}$}, description={new start symbol of the augmented grammar}}
\newglossaryentry{srparser}{name={$\langle Q, q_0, \textsl{action}, \textsl{goto} \rangle$}, description={shift-reduce parser}}
\newglossaryentry{action}{name={$\textsl{action}(q, t)$}, description={action of a shift-reduce parser in state $q$ on token $t$}}
\newglossaryentry{goto}{name={$\textsl{goto}(q, v)$}, description={successor state of $q$ after reducing to the variable $v$}}
\newglossaryentry{marked}{name={$a \rightarrow \beta \bullet \gamma$}, description={marked rule}}
\newglossaryentry{closure}{name={$\textsl{closure}(\mathcal{M})$}, description={closure of the set of marked rules $\mathcal{M}$}}
\newglossaryentry{dollar}{name={\texttt{\$}}, description={end-of-input token}}
\newglossaryentry{gotoM}{name={$\textsl{goto}(\mathcal{M}, c)$}, description={successor state of $\mathcal{M}$ after reading the symbol $c$}}
\newglossaryentry{first}{name={$\textsl{First}(\alpha)$}, description={tokens that can start a string derived from $\alpha$}}
\newglossaryentry{follow}{name={$\textsl{Follow}(a)$}, description={tokens that can follow the variable $a$}}
\newglossaryentry{extmarked}{name={$a \rightarrow \beta \bullet \gamma: L$}, description={extended marked rule with the follow tokens $L$}}
\newglossaryentry{uplus}{name={$\mathcal{M} \uplus \mathcal{N}$}, description={extended union of two sets of extended marked rules}}
